            ],
        ],
    ],{% endif %}
];

Theme Usage

PHP

When {{ page.theme_name }} is configured as your default theme, you can call methods directly: 

flash()->success('Your changes have been saved!');

To use this theme for just this notification (when another theme is your default): 

flash()->use('{{ page.theme_name }}')->success('This uses {{ page.title }}');
JavaScript

When {{ page.theme_name }} is configured as your default theme: 

flasher.success('Your changes have been saved!');

To use this theme for a specific notification: 

flasher.use('{{ page.theme_name }}').success('This uses {{ page.title }}');
