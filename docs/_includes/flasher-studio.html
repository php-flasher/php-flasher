namespace App\Http\Controllers;

use App\Models\Product;
use App\Http\Requests\ProductRequest;
use Illuminate\Http\Request;
use Illuminate\Support\Facades\DB;
use Illuminate\Support\Facades\Log;

class ProductController extends Controller
{
    /**
     * Display a listing of products
     */
    public function index()
    {
        $products = Product::latest()->paginate(10);
        return view('products.index', compact('products'));
    }

    /**
     * Show form for creating a new product
     */
    public function create()
    {
        return view('products.create');
    }

    /**
     * Store a newly created product
     */
    public function store(ProductRequest $request)
    {
        try {
            // Begin transaction
            DB::beginTransaction();

            // Create new product
            $product = new Product;
            $product->name = $request->name;
            $product->description = $request->description;
            $product->price = $request->price;
            $product->category_id = $request->category_id;
            $product->save();

            // Handle image uploads if any
            if ($request->hasFile('image')) {
                $product->addMediaFromRequest('image')
                    ->toMediaCollection('products');
            }

            DB::commit();

            flash()->success('Your product has been created successfully!');

            return redirect()->route('products.index');

        } catch (\Exception $e) {
            DB::rollBack();
            Log::error('Failed to create product: ' . $e->getMessage());

            flash()->error('Failed to create product. Please try again.');

            return back()->withInput();
        }
    }

    /**
     * Show a specific product
     */
    public function show(Product $product)
    {
        return view('products.show', compact('product'));
    }

    /**
     * Edit a specific product
     */
    public function edit(Product $product)
    {
        return view('products.edit', compact('product'));
    }
}

namespace App\Controller;

use App\Entity\Product;
use App\Form\ProductType;
use App\Repository\ProductRepository;
use Doctrine\ORM\EntityManagerInterface;
use Symfony\Bundle\FrameworkBundle\Controller\AbstractController;
use Symfony\Component\HttpFoundation\Request;
use Symfony\Component\HttpFoundation\Response;
use Symfony\Component\Routing\Attribute\Route;

/**
 * @Route("/product", name="product_")
 */
class ProductController extends AbstractController
{
    private $entityManager;
    private $productRepository;

    public function __construct(
        EntityManagerInterface $entityManager,
        ProductRepository $productRepository
    ) {
        $this->entityManager = $entityManager;
        $this->productRepository = $productRepository;
    }

    /**
     * @Route("/", name="index", methods={"GET"})
     */
    public function index(): Response
    {
        $products = $this->productRepository->findAll();

        return $this->render('product/index.html.twig', [
            'products' => $products,
        ]);
    }

    /**
     * @Route("/new", name="new", methods={"GET","POST"})
     */
    public function new(Request $request): Response
    {
        $product = new Product();
        $form = $this->createForm(ProductType::class, $product);
        $form->handleRequest($request);

        if ($form->isSubmitted() && $form->isValid()) {
            try {
                $this->entityManager->persist($product);
                $this->entityManager->flush();

                $this->flash()->success('Your product has been created successfully!');

                return $this->redirectToRoute('product_index');
            } catch (\Exception $e) {
                $this->flash()->error('Failed to create product. Please try again.');
            }
        }

        return $this->render('product/new.html.twig', [
            'product' => $product,
            'form' => $form->createView(),
        ]);
    }

    /**
     * @Route("/{id}", name="show", methods={"GET"})
     */
    public function show(Product $product): Response
    {
        return $this->render('product/show.html.twig', [
            'product' => $product,
        ]);
    }
}

import axios from 'axios';
import flasher from '@flasher/flasher';

class ProductService {
    constructor() {
        this.apiUrl = '/api/products';
    }

    /**
     * Fetch all products
     * @returns {Promise} - Returns products data
     */
    async getAllProducts() {
        try {
            const response = await axios.get(this.apiUrl);
            return response.data;
        } catch (error) {
            this.handleError(error);
            throw error;
        }
    }

    /**
     * Create a new product
     * @param {Object} productData - The product data
     * @returns {Promise} - Returns the created product
     */
    async createProduct(productData) {
        try {
            const response = await axios.post(this.apiUrl, productData);

            flasher.success('Your product has been created successfully!');

            return response.data;
        } catch (error) {
            this.handleError(error);
            throw error;
        }
    }

    /**
     * Update an existing product
     * @param {string} id - Product ID
     * @param {Object} productData - Updated product data
     * @returns {Promise} - Returns the updated product
     */
    async updateProduct(id, productData) {
        try {
            const response = await axios.put(`${this.apiUrl}/${id}`, productData);

            flasher.info('Product has been updated.');

            return response.data;
        } catch (error) {
            this.handleError(error);
            throw error;
        }
    }

    /**
     * Delete a product
     * @param {string} id - Product ID to delete
     * @returns {Promise} - Returns success status
     */
    async deleteProduct(id) {
        try {
            await axios.delete(`${this.apiUrl}/${id}`);

            flasher.warning('Product has been removed.');

            return true;
        } catch (error) {
            this.handleError(error);
            throw error;
        }
    }

    /**
     * Handle errors from API calls
     * @param {Error} error - The error object
     */
    handleError(error) {
        const errorMessage = error.response?.data?.message || 'An unexpected error occurred';

        flasher
            .options({
                position: 'bottom-right',
                timeout: 8000,
                closeButton: true
            })
            .error(errorMessage);

        console.error('API Error:', error);
    }
}

export default new ProductService();
